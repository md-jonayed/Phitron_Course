1. create a django project 
2. create templates and static folder inside that project
3. create base.html file inside the templates folder.
4. store the templates folder inside project's settings.py 
TEMPLATES = [
    {
        'BACKEND': 'django.template.backends.django.DjangoTemplates',
        'DIRS': ['templates'],
        'APP_DIRS': True,
        'OPTIONS': {
            'context_processors': [
                'django.template.context_processors.request',
                'django.contrib.auth.context_processors.auth',
                'django.contrib.messages.context_processors.messages',
            ],
        },
    },
]
5. store the static folder inside project's settings.py 
STATIC_URL = 'static/'

STATICFILES_DIRS = [
    BASE_DIR/'static',
]

6. create views.py inside the project folder 

from django.shortcuts import render


def home(request):
    return render(request, 'base.html')
